287	def check_text_duplicate(new_content, queue_path):
288	    """
289	    최근 5일 이내 발행 완료된 인스타그램 피드 제목(lesson 필드)과
290	    새로 생성된 콘텐츠 제목의 중복 여부를 대조합니다.
291	    """
292	    if not os.path.exists(queue_path):
293	        return False
294	    try:
295	        with open(queue_path, "r", encoding="utf-8") as f:
296	            data = json.load(f)
297	        completed = data.get("completed_lessons", [])
298	        today = datetime.date.today()
299	        five_days_ago = today - datetime.timedelta(days=5)
300	        
301	        past_titles = []
302	        for entry in completed:
303	            if entry.get("subject") != "인스타그램":
304	                continue
305	            entry_date_str = entry.get("date", "")
306	            if not entry_date_str:
307	                continue
308	            try:
309	                entry_date = datetime.datetime.strptime(entry_date_str, "%Y-%m-%d").date()
310	                if entry_date >= five_days_ago:
311	                    lesson_name = entry.get("lesson", "")
312	                    clean_title = lesson_name.replace(".md", "").strip()
313	                    if clean_title:
314	                        past_titles.append(clean_title)
315	            except ValueError:
316	                continue
317	        if not past_titles:
318	            return False
319	            
320	        new_titles = []
321	        for k in ["morning_single", "evening_single", "single"]:
322	            if k in new_content and "title" in new_content[k]:
323	                new_titles.append(new_content[k]["title"].replace("\n", " ").strip())
324	        if "slideshow" in new_content:
325	            for slide in new_content["slideshow"]:
326	                if "title" in slide:
327	                    new_titles.append(slide["title"].replace("\n", " ").strip())
328	                    
329	        for nt in new_titles:
330	            clean_nt = nt.replace(" ", "")
331	            for pt in past_titles:
332	                clean_pt = pt.replace(" ", "")
333	                if clean_nt == clean_pt:
334	                    print(f"[QC-TEXT] Duplicate detected! New: '{nt}' matches Past: '{pt}'")
335	                    return True
336	                # 7자 이상 서브스트링 겹침 검사
337	                for i in range(len(clean_nt) - 6):
338	                    sub = clean_nt[i:i+7]
339	                    if sub in clean_pt:
340	                        print(f"[QC-TEXT] Substring overlap detected! Sub: '{sub}' found in Past: '{pt}'")
341	                        return True
342	        return False
343	    except Exception as e:
344	        print(f"[WARN] Text duplicate check failed: {e}")
345	        return False
