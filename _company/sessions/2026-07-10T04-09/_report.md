def get_current_time():
    return datetime.now()

def calculate_time_to_finish(current_time, minutes=30):
    time_diff = (current_time - datetime.now()).total_seconds() / 60
    remaining_minutes = int(time_diff)
    # 오르간cycles가 completion시에는 끝난다면, 다음cycle에 시작하는 것을 고려합니다.
    if remaining_minutes < 0:
        time_diff += minutes * 24 * 60

    return max(0, time_diff)

def split_agent_task(tasks):
    agents = {}
    for task in tasks:
        agent_id = task['agent_id']
        agent_name = task['name']
        # 해당agent가 가장 가치 있는 task를 결정합니다.
        if agent_id not in agents or get_current_time() - agents[agent_id]['last_activity'] > timedelta(minutes=15):
            agents[agent_id] = {
                'task': task,
                'goal': task['goal'],
                'last_activity': datetime.now()
            }
    return agents

def run_agent_task(agents, tasks):
    for agent, data in agents.items():
        task_id = data['task']['id']
        # 해당agent가 task를 완료할 때의 시간
        completion_time = get_current_time() + timedelta(minutes=calculate_time_to_finish(data['last_activity']))

        print(f"Agent {agent_name}가 Task {task_id}을 완료할 때까지 {completion_time - data['last_activity']}분 지연가 있습니다.")

def find_best_agent(tasks):
    best_agent = None
    min_completion_time = float('inf')
    for agent, task in tasks.items():
        completion_time = get_current_time() + timedelta(minutes=calculate_time_to_finish(task['last_activity']))
        if completion_time < min_completion_time:
            min_completion_time = completion_time
            best_agent = agent
    return best_agent

def split_task_into_small_tasks(tasks):
    small_tasks = []
    for task in tasks:
        small_task_id = task['id']
        data = {
            'task': task,
            'goal': task['goal'],
            'last_activity': datetime.now()
        }
        if not small_task_id:
            # task가 null이면, 다른 agent에게 나누어줍니다
            best_agent = find_best_agent([data])
            tasks.append(data)
            small_tasks.append({'task_id': small_task_id, **data})
    return small_tasks
